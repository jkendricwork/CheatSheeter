GIT & GITHUB QUICK REFERENCE

INITIAL SETUP

# Create repo on GitHub first
cd /path/to/project
git init
git config --global user.name "jkendricwork"
git config --global user.email "jeremy.kendrick@hsc.utah.edu"

CREATE REPO (GH CLI)

# First time: authenticate
gh auth login
# Create repo from current folder
gh repo create --public --source=. --remote=origin
# Or create + push in one command
gh repo create repo-name --public --source=. --push
# Create private repo
gh repo create repo-name --private --source=. --push

BASIC WORKFLOW

git add . # Stage all
git add file.txt # Stage specific
git commit -m "Message"
# First push
git remote add origin https://github.com/user/repo.git
git branch -M main
git push -u origin main
git push # Subsequent pushes

COMMON COMMANDS

git status
git log --oneline
git diff
git pull
git clone https://github.com/user/repo.git

LIST COMMITS

git log # Full history
git log --oneline # Compact
git log --graph --oneline # Visual
git log -n 10 # Last 10
git log --author="jkendricwork"
git log --since="2 weeks ago"

BRANCHING

git branch feature-name
git checkout feature-name
git checkout -b feature-name
git merge feature-name
git branch -d feature-name

UNDO CHANGES

git restore file.txt # Discard changes
git restore --staged file.txt
git reset --soft HEAD~1 # Keep changes
git reset --hard HEAD~1 # Discard all

RESTORE TO COMMIT

git log --oneline # Find hash
git checkout a1b2c3d # View old
git checkout main # Return
# Restore file from commit
git checkout a1b2c3d -- file.txt

RESET TO COMMIT

# Soft reset (keep changes staged)
git reset --soft a1b2c3d
# Mixed reset (keep changes unstaged)
git reset --mixed a1b2c3d
# Hard reset (discard all changes)
git reset --hard a1b2c3d
git push --force # If already pushed
# Revert (safer - new commit)
git revert a1b2c3d
git revert HEAD~3..HEAD # Last 3

.GITIGNORE SETUP

echo "node_modules/" > .gitignore
echo "*.log" >> .gitignore
echo ".env" >> .gitignore
echo "__pycache__/" >> .gitignore
git add .gitignore

FILE HISTORY

git log -- file.txt
git log -p file.txt # With diffs
git blame file.txt # Line by line

STASH (TEMPORARY SAVE)

git stash # Save changes
git stash list
git stash pop # Restore latest
git stash apply stash@{0}
git stash drop

BEST PRACTICES

  • Commits: Small, logical changes with clear messages (present tense: "Add feature" not "Added")
  • Before Push: Always pull first to avoid conflicts
  • Branches: Use for features/experiments, keep main stable
  • Security: Never commit passwords, API keys, or sensitive data
  • Collaboration: Use pull requests, delete merged branches
  • Regular Sync: Pull frequently from remote

⚠️ SAFETY TIPS

  • Use git revert instead of reset for published commits
  • Create backup before destructive ops: git branch backup
  • Never --force on shared branches without coordination
  • Test in separate branch before resetting main
